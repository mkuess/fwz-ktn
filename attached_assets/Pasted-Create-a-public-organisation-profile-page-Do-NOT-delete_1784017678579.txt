Create a public organisation profile page. Do NOT delete the database.

STEP 1 - Create OrganisationController with show method:
app/Http/Controllers/OrganisationController.php

public function show($id) {
    $organisation = \App\Models\Organisation::where('id', $id)
        ->where('is_approved', true)
        ->where('is_active', true)
        ->with(['categories', 'volunteerListings' => function($q) {
            $q->where('is_active', true)->orderBy('created_at', 'desc');
        }])
        ->firstOrFail();
    return view('organisations.show', compact('organisation'));
}

STEP 2 - Add route in routes/web.php:
Route::get('/vereine/{id}', [OrganisationController::class, 'show'])->name('organisations.show');
